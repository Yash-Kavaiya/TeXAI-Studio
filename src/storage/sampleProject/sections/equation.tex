\section{The Equation}
\begin{equation}
  E = mc^2
  \label{eq:mass-energy}
\end{equation}

\noindent where:
\begin{itemize}
  \item $E$ --- the energy of the system, measured in joules (J)
  \item $m$ --- the mass of the system, measured in kilograms (kg)
  \item $c$ --- the speed of light in vacuum,
        approximately $2.998 \times 10^{8}\ \text{m/s}$
\end{itemize}
